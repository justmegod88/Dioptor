        const aiStatusText = document.getElementById('ai-status-text');
        const livePixelDisplay = document.getElementById('live-pixel-display');
        const resDistanceCm = document.getElementById('res-distance-cm');
        const resDistanceM = document.getElementById('res-distance-m');
        const targetTypeBadge = document.getElementById('target-type-badge');
        const toastText = document.getElementById('toast-text');
        const aiEngineInfo = document.getElementById('ai-engine-info');

        const realWidthVal = document.getElementById('real-width-val');
        const realWidthSlider = document.getElementById('real-width-slider');
        const focalDisplay = document.getElementById('focal-display');
        const focalSliderVal = document.getElementById('focal-slider-val');
        const focalSlider = document.getElementById('focal-slider');
        const btnCalibrate50cm = document.getElementById('btn-calibrate-50cm');

        const cameraError = document.getElementById('camera-error');
        const btnRestartCam = document.getElementById('btn-restart-cam');

        const infoModal = document.getElementById('info-modal');
        const btnToggleInfo = document.getElementById('btn-toggle-info');
        const btnCloseInfo = document.getElementById('btn-close-info');

        // Application State
        let currentMode = 'glasses'; // 'glasses', 'cell phone', 'cup', 'card'
        let realWidthCm = 14.0;      // Standard Adult Glasses Frame Width ~14cm
        let focalLengthPx = 700;     // Camera Focal Length
        let livePixelWidth = 0;
        let smoothedDistance = 0;

        // Models
        let cocoModel = null;
        let faceMeshModel = null;
        let cameraUtils = null;
        let isProcessingFrame = false;

        async function initAIModels() {
            aiStatusText.textContent = "AI 모델 로딩 중 (COCO-SSD & MediaPipe)...";
            try {
                // 1. Load TensorFlow COCO-SSD Object Detector
                cocoModel = await cocoSsd.load();

                // 2. Load MediaPipe Face Mesh for Glasses Temple Landmark Tracking
                faceMeshModel = new FaceMesh({
                    locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/face_mesh/${file}`
                });
                faceMeshModel.setOptions({
                    maxNumFaces: 1,
                    refineLandmarks: true,
                    minDetectionConfidence: 0.5,
                    minTrackingConfidence: 0.5
                });
                faceMeshModel.onResults(onFaceMeshResults);

                aiStatusDot.className = "w-2.5 h-2.5 rounded-full bg-emerald-400 animate-pulse";
                aiStatusText.textContent = "AI 모델 준비 완료";
            } catch (err) {
                console.error("AI Model Load Error:", err);
                aiStatusDot.className = "w-2.5 h-2.5 rounded-full bg-amber-500";
                aiStatusText.textContent = "기본 추적 모드로 실행";
            }
        }

        async function startCamera() {
            cameraError.classList.add('hidden');
            try {
                cameraUtils = new Camera(webcam, {
                    onFrame: async () => {
                        if (isProcessingFrame) return;
                        isProcessingFrame = true;

                        if (currentMode === 'glasses' && faceMeshModel) {
                            // Try MediaPipe Face Mesh first for worn glasses frame
                            await faceMeshModel.send({ image: webcam });
                        } else {
                            // Use COCO-SSD / Color BBox Tracker for general items
                            await processGeneralObjectFrame();
                        }
                        isProcessingFrame = false;
                    },
                    width: 1280,
                    height: 720
                });

                await cameraUtils.start();
            } catch (err) {
                console.error("Camera Error:", err);
                cameraError.classList.remove('hidden');
            }
        }

        function onFaceMeshResults(results) {
            canvas.width = results.image.width;
            canvas.height = results.image.height;

            ctx.save();
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.drawImage(results.image, 0, 0, canvas.width, canvas.height);

            if (results.multiFaceLandmarks && results.multiFaceLandmarks.length > 0) {
                const landmarks = results.multiFaceLandmarks[0];

                // Glasses Frame Width Landmarks:
                // #127: Left Temple/Cheekbone frame edge | #356: Right Temple/Cheekbone frame edge
                const leftTemple = landmarks[127];
                const rightTemple = landmarks[356];

                const x1 = leftTemple.x * canvas.width;
                const y1 = leftTemple.y * canvas.height;
                const x2 = rightTemple.x * canvas.width;
                const y2 = rightTemple.y * canvas.height;

                const pixelWidth = Math.hypot(x2 - x1, y2 - y1);
                livePixelWidth = pixelWidth;

                // Draw Glasses Frame Landmark Line
                ctx.strokeStyle = '#06b6d4'; // Cyan
                ctx.lineWidth = 4;
                ctx.beginPath();
                ctx.moveTo(x1, y1);
                ctx.lineTo(x2, y2);
                ctx.stroke();

                // Draw Frame End Pins
                ctx.fillStyle = '#38bdf8';
                ctx.beginPath(); ctx.arc(x1, y1, 8, 0, 2 * Math.PI); ctx.fill();
                ctx.beginPath(); ctx.arc(x2, y2, 8, 0, 2 * Math.PI); ctx.fill();

                // Draw Glasses Box Overlay
                const boxPadding = pixelWidth * 0.15;
                const boxX = Math.min(x1, x2) - boxPadding;
                const boxY = Math.min(y1, y2) - pixelWidth * 0.2;
                const boxW = pixelWidth + (boxPadding * 2);
                const boxH = pixelWidth * 0.45;

                ctx.strokeStyle = 'rgba(56, 189, 248, 0.6)';
                ctx.lineWidth = 2;
                ctx.strokeRect(boxX, boxY, boxW, boxH);

                ctx.fillStyle = '#38bdf8';
                ctx.font = 'bold 16px Pretendard';
                ctx.fillText(`👓 안경 감지됨 (${Math.round(pixelWidth)}px)`, boxX, boxY - 8);

                calculateAndRenderDistance(pixelWidth, "안경 테 폭 감지 완료");
            } else {
                // If Face Mesh doesn't find face, fallback to COCO-SSD for standalone glasses in hand
                processGeneralObjectFrame();
            }

            ctx.restore();
        }

        async function processGeneralObjectFrame() {
            canvas.width = webcam.videoWidth || 1280;
            canvas.height = webcam.videoHeight || 720;

            ctx.save();
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.drawImage(webcam, 0, 0, canvas.width, canvas.height);

            if (!cocoModel) {
                ctx.restore();
                return;
            }

            try {
                const predictions = await cocoModel.detect(webcam);
                let targetFound = false;

                // Filter predictions based on current selected mode
                for (let pred of predictions) {
                    const label = pred.class.toLowerCase();
                    const isTargetClass = 
                        (currentMode === 'cell phone' && (label === 'cell phone' || label === 'phone')) ||
                        (currentMode === 'cup' && (label === 'cup' || label === 'bottle' || label === 'wine glass')) ||
                        (currentMode === 'glasses' && (label === 'eyeglasses' || label === 'glasses' || label === 'person')) ||
                        (currentMode === 'card');

                    if (isTargetClass || predictions.length > 0) {
                        const [x, y, width, height] = pred.bbox;
                        livePixelWidth = width;
                        targetFound = true;

                        // Draw Object Bounding Box
                        ctx.strokeStyle = '#38bdf8';
                        ctx.lineWidth = 4;
                        ctx.strokeRect(x, y, width, height);

                        ctx.fillStyle = '#0284c7';
                        ctx.fillRect(x, y - 28, Math.max(120, pred.class.length * 12), 28);

                        ctx.fillStyle = '#ffffff';
                        ctx.font = 'bold 14px Pretendard';
                        ctx.fillText(`${pred.class} (${Math.round(width)}px)`, x + 6, y - 8);

                        calculateAndRenderDistance(width, `${pred.class} 감지됨`);
                        break;
                    }
                }

                if (!targetFound) {
                    // Fallback to center color/contrast bounding box if object model missed card/custom object
                    processColorBoundingBoxFallback();
                }

            } catch (e) {
                console.error("COCO-SSD Error:", e);
            }

            ctx.restore();
        }

        function processColorBoundingBoxFallback() {
            const centerX = canvas.width / 2;
            const centerY = canvas.height / 2;
            const defaultBoxW = canvas.width * 0.25;
            const defaultBoxH = canvas.width * 0.16;

            const boxX = centerX - defaultBoxW / 2;
            const boxY = centerY - defaultBoxH / 2;

            ctx.strokeStyle = 'rgba(234, 179, 8, 0.8)';
            ctx.lineWidth = 3;
            ctx.setLineDash([6, 6]);
            ctx.strokeRect(boxX, boxY, defaultBoxW, defaultBoxH);
            ctx.setLineDash([]);

            ctx.fillStyle = '#eab308';
            ctx.font = 'bold 13px Pretendard';
            ctx.fillText("가이드 상자 안에 사물을 맞춰주세요", boxX, boxY - 8);

            livePixelWidth = defaultBoxW;
            calculateAndRenderDistance(defaultBoxW, "가이드 박스 기반 거리 측정 중");
        }

        function calculateAndRenderDistance(pixelDist, msg) {
            livePixelDisplay.textContent = Math.round(pixelDist);

            if (pixelDist > 5) {
                // Distance formula: d = (W * f) / p
                const rawDistanceCm = (realWidthCm * focalLengthPx) / pixelDist;

                if (smoothedDistance === 0) smoothedDistance = rawDistanceCm;
                else smoothedDistance = smoothedDistance * 0.75 + rawDistanceCm * 0.25;

                resDistanceCm.textContent = smoothedDistance.toFixed(1);
                resDistanceM.textContent = `(${(smoothedDistance / 100).toFixed(2)} m)`;

                toastText.textContent = `${msg} (픽셀: ${Math.round(pixelDist)}px)`;
            }
        }

        const objButtons = document.querySelectorAll('.obj-btn');

                objButtons.forEach(b => {
                    b.className = "obj-btn p-3 rounded-2xl border border-gray-800 bg-gray-950 text-gray-400 hover:text-white text-xs font-bold transition flex flex-col items-center gap-1.5";
                });

                btn.className = "obj-btn active p-3 rounded-2xl border border-cyan-500 bg-cyan-500/10 text-cyan-300 text-xs font-bold transition flex flex-col items-center gap-1.5";
                currentMode = btn.getAttribute('data-mode');

                    targetTypeBadge.textContent = "Glasses Mode";

                    realWidthCm = 7.5;
                    targetTypeBadge.textContent = "Cell Phone Mode";

                    aiEngineInfo.textContent = "스마트 경계 박스 트래커";

                realWidthVal.textContent = `${realWidthCm.toFixed(1)} cm`;
